"""tiny_twin.py — baseline "original tiny-twin" simulator (v1).
A deliberately simple, lazy lumped RC (resistor-capacitor) single-zone thermal
twin. It steps a first-order envelope ODE at the dataset resolution (10
minutes) with explicit thermostat-setpoint heating.
All coefficients are FIXED building-handbook defaults (module constants).
The model never reads, fits, or otherwise touches the real data in any form:
it is the uncalibrated baseline that the new calibrated twin
(tiny_twin_v2.py) is compared against.
Pure Python standard library, zero dependencies.
"""
from __future__ import annotations
T_SET = 21.0      # thermostat heating setpoint (deg C)
A_ENV = 0.02      # envelope coupling: per-step fraction of (T_out - T)
G_SET = 0.02      # heating loop gain toward T_SET when T < T_SET
A_NRG = 0.01      # appliance heat gain (per 100-Wh load step)
Q_SCALE = 100.0   # appliance energy reported in Wh/10min; normalize to 100-Wh units
def step(
    temp: float,
    t_out: float,
    load_wh: float,
    t_set: float = T_SET,
    a_env: float = A_ENV,
    g_set: float = G_SET,
    a_nrg: float = A_NRG,
    q_scale: float = Q_SCALE,
) -> float:
    """One discrete 10-min step of the baseline twin.
    Returns the room temperature at the next sample:
        T += a_env*(T_out - T) + g_set*max(T_SET - T, 0) + a_nrg*(load/100)
    """
    heating = max(t_set - temp, 0.0)  # clamped thermostat: only heat when below setpoint
    d_temp = a_env * (t_out - temp) + g_set * heating + a_nrg * (load_wh / q_scale)
    return temp + d_temp

    def __init__(self, alpha: float = 0.0) -> None:

    def fit(self, x_rows: list[list[float]], y: list[float]) -> None:
        n = len(y)
        ncol = len(x_rows[0]) if n else 0
        self.mean = [fmean(r[i] for r in x_rows) for i in range(ncol)]
        sds = []
        for i in range(ncol):

            sds.append(s if s > 1e-12 else 1.0)

def _solve_gauss(a: list[list[float]], b: list[float]) -> list[float]:
    """Solve A x = b (square system) by plain Gaussian elimination."""
    n = len(b)
    a = [row[:] for row in a]
    b = b[:]
    for col in range(n):
        pivot = max(range(col, n), key=lambda r: abs(a[r][col]))
        if abs(a[pivot][col]) < 1e-12:
            a[col][col] = 1.0  # degenerate column -> treat as zero-dim
            continue
        if pivot != col:
            a[col], a[pivot] = a[pivot], a[col]
            b[col], b[pivot] = b[pivot], b[col]
        inv = 1.0 / a[col][col]
        for i in range(col, n):
            a[col][i] *= inv
        b[col] *= inv
        for r in range(n):
            if r != col:
                f = a[r][col]
                if f != 0.0:
                    for i in range(col, n):
                        a[r][i] -= f * a[col][i]
                    b[r] -= f * b[col]
    return b

"""cmri_twin.py — the cmri-tiny-twin: a more realistic, physically-motivated twin.
Beats the single-zone v1/v2 twins by adding the physics the first-order lumped
RC model ignores:
  1. TWO-ZONE COUPLING.  The kitchen (T1) exchanges heat with the rest of the
     house (T_h = mean of T2,T3,T4,T5,T7,T8,T9 — T6 excluded: an anomalous
     sensor, min -6.07 C, mean 7.9 C vs 19-22 C for every other zone sensor,
     consistent with an unheated/outlying space).  Both zones are simulated
     states coupled through (T_h - T1) / (T1 - T_h) conductance terms.
  2. WIND-DRIVEN INFILTRATION.  Air leakage scales with wind speed:
     wind * (T_out - T1) — mechanically distinct from the still-air envelope
     term (T_out - T1).
  3. CLEAR-SKY SOLAR GAIN.  A physical solar-irradiance proxy (clear-sky GHI,
     W/m^2) computed for the house location (Chievres, Belgium: 50.56 N,
     3.83 E), DST-aware (Belgium moved from UTC+1 to UTC+2 on 2016-03-27,
     inside this dataset), with solar-elevation geometry, equation-of-time and
     Kasten-Young air mass.
  4. FABRIC THERMAL MASS.  A building-fabric state T_mass (the internal mass
     of walls/floors, which equilibrates with the outside air over many
     hours, not minutes) as an exogenous EMA of T_out: with dt = 600 s and
     time constant tau, T_mass[t] = (1-w)*T_out[t] + w*T_mass[t-1],
     w = exp(-dt/tau).  Because first-order RC zones already track short
     time scales through the (T_out - T) term, the slow T_mass drives the
     long-horizon open-loop trend.  T_mass is a pure function of observed
     weather, so the step API stays single-step/exogenous (no model state
     to re-anchor).
     TWO-TIMESCALE FABRIC (driver 9).  Experiments show the open-loop drift
     on the held-out block is a slow seasonal level shift between the
     calibration and evaluation seasons.  A single fabric capacitor (driver
     8, tau = 3 h) tracks weather swings but not the seasonal level.  Driver
     9 adds a second, SLOWER capacitor T_mass2 (tau of hours-to-days) to the
     same driver set, giving the open-loop twin a slow exogenous reference
     for the seasonal level: dT now includes a9*(T_mass-T) + a10*(T_mass2-T).

def _last_sunday(year: int, month: int) -> dt.date:
    nd = dt.date(year, month + 1, 1) - dt.timedelta(days=1) if month < 12 else dt.date(year, 12, 31)
    return nd - dt.timedelta(days=(nd.weekday() - 6) % 7)
def utc_offset_hours(d: dt.datetime) -> float:
    """Europe/Brussels: UTC+1 winter, UTC+2 summer (last Sun Mar -> last Sun Oct)."""
    start = dt.datetime.combine(_last_sunday(d.year, 3), dt.time(1, 0))
    end = dt.datetime.combine(_last_sunday(d.year, 10), dt.time(1, 0))
    return 2.0 if start <= d < end else 1.0
def clear_sky_ghi(d: dt.datetime) -> float:
    """Clear-sky horizontal irradiance (W/m^2) at the house location.
    Direct-beam + diffuse approximation using solar-elevation geometry and the
    Kasten-Young air mass.  Units are physical (0-900 W/m^2) so the fitted
    solar coefficient has a physical interpretation (C per W/m^2 per 10 min).
    """
    n = d.timetuple().tm_yday
    utc = d - dt.timedelta(hours=utc_offset_hours(d))
    decl = math.radians(23.44 * math.sin(math.radians(360.0 * (n - 81) / 365)))
    B = math.radians(360.0 * (n - 81) / 364)
    eot_min = 9.87 * math.sin(2 * B) - 7.53 * math.cos(B) - 1.5 * math.sin(B)
    solar_noon_utc_h = 12.0 - LON_DEG / 15.0
    utc_h = utc.hour + utc.minute / 60.0 + utc.second / 3600.0 + eot_min / 60.0
    H = math.radians((solar_noon_utc_h - utc_h) * 15.0)
    sin_el = (math.sin(math.radians(LAT_DEG)) * math.sin(decl)
              + math.cos(math.radians(LAT_DEG)) * math.cos(decl) * math.cos(H))
    if sin_el <= 0.0:
        return 0.0
    el_deg = math.degrees(math.asin(sin_el))
    am = 1.0 / (sin_el + 0.50572 * (el_deg + 6.07995) ** -1.6364)
    i0 = 1361.0 * (1.0 + 0.033 * math.cos(math.radians(360.0 * (n - 2) / 365)))
    return 0.9 * i0 * sin_el * 0.7 ** (am ** 0.678)

def metrics(pred, obs):
    n = len(obs)
    mae = sum(abs(p - o) for p, o in zip(pred, obs)) / n
    rmse = math.sqrt(sum((p - o) ** 2 for p, o in zip(pred, obs)) / n)
    mape = 100.0 * sum(abs(p - o) / abs(o) for p, o in zip(pred, obs)) / n
    ybar = st.fmean(obs)
    ss_res = sum((p - o) ** 2 for p, o in zip(pred, obs))
    ss_tot = sum((o - ybar) ** 2 for o in obs)
    r2 = 1.0 - ss_res / ss_tot if ss_tot > 0 else 0.0
    return {"rmse": rmse, "mae": mae, "r2": r2, "mape": mape}

def run_protocol_cmri(twin, state, recs, resync_days):
    """Step a two-zone twin over records; return kitchen predictions.
    recs: list of (T1_obs, Th_obs, t_out, Tdewpoint, Appliances, lights,
                   Windspeed, GHI, T_mass) -- optionally a 10th entry T_mass2
                   (slow fabric state) for two-timescale driver-set selection.
    resync_days: list of index-lists (one per day), or None for free-run.
    state: initial observed (T1, Th).

    Returns kitchen pred list of len(recs); None where no prediction exists
    (first sample of each resync day, or sample 0 in free-run).

    n = len(recs)
    pred = [None] * n
    if resync_days is None:
        t1, th = state
        for j in range(1, n):
            r = recs[j - 1]
            _, _, t_out, dew, load, lux, wind, ghi, tm, *rest = r
            t1, th = twin.step(t1, th, t_out, dew, load, lux, wind, ghi, tm,
                               rest[0] if rest else None)
            pred[j] = t1
        return pred
    for day in resync_days:
        t1, th = recs[day[0]][0], recs[day[0]][1]
        for a, b in zip(day[:-1], day[1:]):
            r = recs[a]
            _, _, t_out, dew, load, lux, wind, ghi, tm, *rest = r
            t1, th = twin.step(t1, th, t_out, dew, load, lux, wind, ghi, tm,
                               rest[0] if rest else None)
            pred[b] = t1
    return pred
def days_of(dates, off, end=None):
    """Group indices [off, end) into day-index lists (0-based within block)."""
    end = len(dates) if end is None else end
    groups: dict[dt.date, list[int]] = {}
    for j in range(end - off):
        groups.setdefault(dates[off + j].date(), []).append(j)
    return [v for _, v in sorted(groups.items())]

        self.h.fit(Xh, yh)
    def step(self, t1, th, t_out, dew, load, lux, wind, ghi, t_mass,
             t_mass2=None, t_set=CMRI_T_SET):
        fk = CmriTwin._k_feats(t1, th, t_out, dew, load, lux, wind, ghi,

"""select_cv.py — honest within-train cross-validation for cmri twin knobs.

    """Daily-resync kitchen RMSE (extended state anchored from observations)."""

def pearson(a: list[float], b: list[float]) -> float:
    na, nb = len(a), len(b)
    if na < 2 or nb < 2:
        return float("nan")

    sxy = sum((x - ma) * (y - mb) for x, y in zip(a, b))

def dm_hac(d: list[float], bw: int) -> tuple[float, float]:
    """Diebold-Mariano with Newey-West HAC long-run variance.
    The 10-min resync errors are strongly autocorrelated within a day, so the
    naive DM (which treats each step as independent) overstates significance.
    bw is the autocovariance truncation/bandwidth in steps — use one resync
    block (144 steps) plus, conservatively, two blocks.
    """
    n = len(d)
    m = fmean(d)
    g0 = sum((x - m) ** 2 for x in d) / n
    lrv = g0
    for l in range(1, bw + 1):
        gl = sum((d[i] - m) * (d[i + l] - m) for i in range(n - l)) / n
        w = 1.0 - l / (bw + 1.0)  # Bartlett kernel
        lrv += 2.0 * w * gl
    if lrv <= 0.0:
        return math.nan, math.nan
    stat = m / math.sqrt(lrv / n)
    return stat, two_sided_normal_p(stat)

verify_freerun.py (§5.2) showed the free-run edge is *first-half-owned*: over the
same 42-day window the open-loop mean daily d is +0.344 C on days 1-21 (14/21)
and -0.387 C on days 22-42 (5/21).  A reviewer's natural worry is that the

by a sub-period or a few big days, i.e. that the §7 top-episode days (2016-04-22,

directly: halves, thirds, calendar weeks, an OLS trend and rank correlation of

largest-|d| days).  Findings:

def phi(z: float) -> float:
    """Standard normal CDF (via erff, pure stdlib)."""
    return 0.5 * (1.0 + math.erf(z / math.sqrt(2.0)))
def sign_test(wins: int, n: int) -> tuple[float, float]:
    """Binomial sign test on counts of positive paired d, normal approx."""
    mu = n / 2.0
    sd = math.sqrt(n / 4.0)
    z = (wins - 0.5 - mu) / sd  # continuity correction
    return z, 1.0 - phi(z)      # one-sided H1: P(d>0) > 0.5
def ols(y: list[float]) -> tuple[float, float, float]:
    """OLS slope, se, t-statistic of y on x = 0..n-1 (day index)."""
    n = len(y)
    xs = list(range(n))
    mx, my = statistics.fmean(xs), statistics.fmean(y)
    sxx = sum((x - mx) ** 2 for x in xs)
    sxy = sum((x - mx) * (yy - my) for x, yy in zip(xs, y))
    b = sxy / sxx
    a = my - b * mx
    resid = [yy - (a + b * x) for x, yy in zip(xs, y)]
    se = math.sqrt(sum(r * r for r in resid) / (n - 2) / sxx)
    return b, se, b / se
def spearman(xs: list[int], ys: list[float]) -> float:
    n = len(ys)
    rk = [1 + sum(1 for j in range(n) if ys[j] < ys[i]
                  or (ys[j] == ys[i] and j < i)) for i in range(n)]
    return 1.0 - 6.0 * sum((x - r) ** 2 for x, r in zip(xs, rk)) / (n * (n * n - 1))
